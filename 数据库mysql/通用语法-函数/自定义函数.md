```mysql
delimiter //
create function 你的函数名(参数1 类型, 参数2 类型)
returns 返回类型
deterministic
begin
    declare 变量名 类型;
    set 变量名 = 你的逻辑;
    return 变量名;
end //
delimiter ;
```
